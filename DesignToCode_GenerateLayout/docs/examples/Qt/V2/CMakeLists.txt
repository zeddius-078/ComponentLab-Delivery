cmake_minimum_required(VERSION 3.21)
project(LayoutParserExamples LANGUAGES CXX)
get_filename_component(D2C_PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(D2C "${D2C_PROJECT_ROOT}/Source")
set(D2C_PIPELINE_ROOT "${D2C}/HtmlDsl/pipeline")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Gui Widgets Svg Test UiTools Xml)
include("${D2C}/GeneratedComponentDependency.cmake")
if(MSVC)
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:")
  add_compile_options(/W4 /permissive- /utf-8)
endif()
add_subdirectory("${D2C}" "${CMAKE_BINARY_DIR}/parser")
set_target_properties(v2_ui_compiler PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
add_library(v2_runtime STATIC src/UiPage.cpp src/UiPage.h src/WorkflowWindow.cpp src/WorkflowWindow.h)
target_include_directories(v2_runtime PUBLIC src)
target_link_libraries(v2_runtime PUBLIC v2_widgets Qt5::UiTools)
add_executable(d2c_v2 src/main.cpp)
target_link_libraries(d2c_v2 PRIVATE v2_runtime)
target_compile_definitions(d2c_v2 PRIVATE D2C_DEFAULT_DATA="/data/html-dsl")
add_executable(v2_tests "${D2C_PROJECT_ROOT}/docs/validation/v2/V2Tests.cpp")
target_link_libraries(v2_tests PRIVATE v2_runtime Qt5::Test Qt5::Xml)
target_compile_definitions(v2_tests PRIVATE V2_ROOT="${CMAKE_CURRENT_SOURCE_DIR}" D2C_ROOT="${D2C}"
  PIPELINE_ROOT="${D2C_PIPELINE_ROOT}" FIXTURE_ROOT="${D2C_PROJECT_ROOT}/docs/fixtures/layout")
enable_testing()
file(MAKE_DIRECTORY "${D2C_PROJECT_ROOT}/output/logs/v2")
add_test(NAME v2_workflow COMMAND v2_tests -o "${D2C_PROJECT_ROOT}/output/logs/v2/qt-results.xml,xml")
set_tests_properties(v2_workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
