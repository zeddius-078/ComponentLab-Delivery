# V2 构建提供静态布局生成器和 XML 运行器；组件来自已导入 SDK。
cmake_minimum_required(VERSION 3.21)
project(DesignToCodeV2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Gui Widgets Svg Test UiTools Xml)
set(D2C "${CMAKE_CURRENT_SOURCE_DIR}/../..")
include("${D2C}/GeneratedComponentDependency.cmake")
set(D2C_ROUTE_FOLDER "HtmlDsl" CACHE STRING "HTML + DSL pipeline folder")
set(D2C_PIPELINE_ROOT "${D2C}/${D2C_ROUTE_FOLDER}/pipeline")
# 仅向 MSVC 添加其专用警告、编码与兼容参数，其他编译器保持各自默认配置。
if(MSVC)
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:")
  add_compile_options(/W4 /permissive- /utf-8)
  # 当前新版 MSVC 与 Qt 5.15.2 组合需要兼容头，严格限定版本以免影响其他工具链。
  if(MSVC_VERSION GREATER_EQUAL 1951 AND Qt5Core_VERSION VERSION_EQUAL 5.15.2)
    add_compile_options("/FI${GENERATED_COMPONENT_SDK_ROOT}/include/MyDesignSystem/qt5152_msvc_compat.h")
  endif()
endif()
# 引入布局转换引擎；组件类型只从已发布 SDK 取得。
add_subdirectory("${D2C}" "${CMAKE_BINARY_DIR}/v1")
add_library(v2_widgets STATIC "${D2C_PIPELINE_ROOT}/runtime/WidgetRegistry.cpp" "${D2C_PIPELINE_ROOT}/runtime/WidgetRegistry.h")
target_include_directories(v2_widgets PUBLIC "${D2C_PIPELINE_ROOT}/runtime")
target_link_libraries(v2_widgets PUBLIC GeneratedComponentSDK::Components)
# 构建期生成器链接共享转换引擎；运行库只链接组件工厂和 Qt UiTools。
add_executable(v2_ui_compiler "${D2C_PIPELINE_ROOT}/compiler/SceneBuilder.cpp" "${D2C_PIPELINE_ROOT}/compiler/SceneBuilder.h" "${D2C_PIPELINE_ROOT}/compiler/UiCompiler.cpp")
target_link_libraries(v2_ui_compiler PRIVATE d2c_engines v2_widgets)
add_library(v2_runtime STATIC src/UiPage.cpp src/UiPage.h src/WorkflowWindow.cpp src/WorkflowWindow.h)
target_include_directories(v2_runtime PUBLIC src)
target_link_libraries(v2_runtime PUBLIC v2_widgets Qt5::UiTools)
add_executable(d2c_v2 src/main.cpp)
target_link_libraries(d2c_v2 PRIVATE v2_runtime)
# 运行器只加载 HTML + DSL 生成的静态页面包。
target_compile_definitions(d2c_v2 PRIVATE D2C_DEFAULT_DATA="/data/html-dsl")
# 测试覆盖 XML 加载、状态更新、重复生成及设计节点对应，报告集中到验证目录。
add_executable(v2_tests "${D2C}/Validation/v2/V2Tests.cpp")
target_link_libraries(v2_tests PRIVATE v2_runtime Qt5::Test Qt5::Xml)
target_compile_definitions(v2_tests PRIVATE V2_ROOT="${CMAKE_CURRENT_SOURCE_DIR}" D2C_ROOT="${D2C}" PIPELINE_ROOT="${D2C_PIPELINE_ROOT}")
enable_testing()
add_test(NAME v2_workflow COMMAND v2_tests -o "${D2C}/../build/reports/v2/qt-results.xml,xml")
set_tests_properties(v2_workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
