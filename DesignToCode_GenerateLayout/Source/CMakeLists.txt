# V1 原生构建：组件库 → 共享转换引擎 → 预览程序与测试。
cmake_minimum_required(VERSION 3.21)
project(DesignToCodeV1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Gui Widgets Svg Test)
include("${CMAKE_CURRENT_SOURCE_DIR}/GeneratedComponentDependency.cmake")
# 仅向 MSVC 添加其专用警告、编码与兼容参数，其他编译器保持各自默认配置。
if(MSVC)
  # UTF-8 支持中文源码；兼容头仅针对当前新版 MSVC 与 Qt 5.15.2 组合。
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:")
  add_compile_options(/W4 /permissive- /utf-8)
  # 当前新版 MSVC 与 Qt 5.15.2 组合需要兼容头，严格限定版本以免影响其他工具链。
  if(MSVC_VERSION GREATER_EQUAL 1951 AND Qt5Core_VERSION VERSION_EQUAL 5.15.2)
    add_compile_options("/FI${GENERATED_COMPONENT_SDK_ROOT}/include/MyDesignSystem/qt5152_msvc_compat.h")
  endif()
endif()
# 转换引擎固定读取唯一的 HTML + DSL 路线。
if(NOT D2C_PIPELINE_ROOT)
  set(D2C_PIPELINE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/HtmlDsl/pipeline")
endif()
# 引擎只依赖组件库；两类测试分别覆盖组件映射和几何布局。
add_library(d2c_engines STATIC ${D2C_PIPELINE_ROOT}/engines/component_converter/ComponentConverter.cpp ${D2C_PIPELINE_ROOT}/engines/component_converter/ComponentConverter.h)
target_include_directories(d2c_engines PUBLIC ${D2C_PIPELINE_ROOT}/engines/component_converter)
target_link_libraries(d2c_engines PUBLIC GeneratedComponentSDK::Components)
add_executable(component_tests Validation/component/ComponentTests.cpp)
target_link_libraries(component_tests PRIVATE d2c_engines Qt5::Test)
target_compile_definitions(component_tests PRIVATE D2C_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
# XML 测试结果集中写到 Validation，使用 offscreen 平台运行。
enable_testing()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../build/reports/component" "${CMAKE_CURRENT_SOURCE_DIR}/../build/reports/layout")
add_test(NAME component_converter COMMAND component_tests -o "${CMAKE_CURRENT_SOURCE_DIR}/../build/reports/component/qt-results.xml,xml")
set_tests_properties(component_converter PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
target_sources(d2c_engines PRIVATE ${D2C_PIPELINE_ROOT}/engines/layout_converter/LayoutConverter.cpp ${D2C_PIPELINE_ROOT}/engines/layout_converter/LayoutConverter.h)
target_include_directories(d2c_engines PUBLIC ${D2C_PIPELINE_ROOT}/engines/layout_converter)
add_executable(layout_tests Validation/layout/LayoutTests.cpp)
target_compile_definitions(layout_tests PRIVATE D2C_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(layout_tests PRIVATE d2c_engines Qt5::Test)
add_test(NAME layout_converter COMMAND layout_tests -o "${CMAKE_CURRENT_SOURCE_DIR}/../build/reports/layout/qt-results.xml,xml")
set_tests_properties(layout_converter PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(d2c_preview Qt/V1/main.cpp)
target_link_libraries(d2c_preview PRIVATE d2c_engines)
