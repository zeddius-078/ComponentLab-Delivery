cmake_minimum_required(VERSION 3.21)
project(LayoutParser LANGUAGES CXX)
get_filename_component(D2C_PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
file(REAL_PATH "${CMAKE_BINARY_DIR}" _binary)
file(REAL_PATH "${D2C_PROJECT_ROOT}/build" _build)
cmake_path(IS_PREFIX _build "${_binary}" NORMALIZE _in_build)
if(NOT _in_build)
  message(FATAL_ERROR "Build outputs must stay inside ${D2C_PROJECT_ROOT}/build")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Gui Widgets Svg Test)
include("${CMAKE_CURRENT_SOURCE_DIR}/GeneratedComponentDependency.cmake")
if(MSVC)
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:")
  add_compile_options(/W4 /permissive- /utf-8)
  if(MSVC_VERSION GREATER_EQUAL 1951 AND Qt5Core_VERSION VERSION_EQUAL 5.15.2)
    add_compile_options("/FI${GENERATED_COMPONENT_SDK_ROOT}/include/MyDesignSystem/qt5152_msvc_compat.h")
  endif()
endif()
set(D2C_PIPELINE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/HtmlDsl/pipeline")
add_library(d2c_engines STATIC
  "${D2C_PIPELINE_ROOT}/engines/component_converter/ComponentConverter.cpp"
  "${D2C_PIPELINE_ROOT}/engines/layout_converter/LayoutConverter.cpp")
target_include_directories(d2c_engines PUBLIC
  "${D2C_PIPELINE_ROOT}/engines/component_converter"
  "${D2C_PIPELINE_ROOT}/engines/layout_converter")
target_link_libraries(d2c_engines PUBLIC GeneratedComponentSDK::Components)
add_library(v2_widgets STATIC "${D2C_PIPELINE_ROOT}/runtime/WidgetRegistry.cpp")
target_include_directories(v2_widgets PUBLIC "${D2C_PIPELINE_ROOT}/runtime")
target_link_libraries(v2_widgets PUBLIC GeneratedComponentSDK::Components)
add_executable(v2_ui_compiler
  "${D2C_PIPELINE_ROOT}/compiler/SceneBuilder.cpp"
  "${D2C_PIPELINE_ROOT}/compiler/UiCompiler.cpp")
target_link_libraries(v2_ui_compiler PRIVATE d2c_engines v2_widgets)
target_compile_definitions(v2_ui_compiler PRIVATE D2C_PROJECT_ROOT="${D2C_PROJECT_ROOT}")
add_custom_target(layout_generator_demo DEPENDS v2_ui_compiler)
option(D2C_BUILD_TESTS "Build parser regression tests maintained in docs" OFF)
option(D2C_BUILD_PREVIEW "Build the documented V1 example" OFF)
if(D2C_BUILD_TESTS)
  enable_testing()
  foreach(group component layout)
    if(group STREQUAL "component")
      set(test_source ComponentTests.cpp)
    else()
      set(test_source LayoutTests.cpp)
    endif()
    add_executable(${group}_tests "${D2C_PROJECT_ROOT}/docs/validation/${group}/${test_source}")
    target_link_libraries(${group}_tests PRIVATE d2c_engines Qt5::Test)
    target_compile_definitions(${group}_tests PRIVATE D2C_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
    file(MAKE_DIRECTORY "${D2C_PROJECT_ROOT}/build/reports/${group}")
    add_test(NAME ${group}_converter COMMAND ${group}_tests
      -o "${D2C_PROJECT_ROOT}/build/reports/${group}/qt-results.xml,xml")
    set_tests_properties(${group}_converter PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endforeach()
endif()
if(D2C_BUILD_PREVIEW)
  add_executable(d2c_preview "${D2C_PROJECT_ROOT}/docs/examples/Qt/V1/main.cpp")
  target_link_libraries(d2c_preview PRIVATE d2c_engines)
endif()
