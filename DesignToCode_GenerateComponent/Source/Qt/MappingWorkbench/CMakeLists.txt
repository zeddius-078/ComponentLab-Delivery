cmake_minimum_required(VERSION 3.21)
project(DesignToCodeWorkbench LANGUAGES CXX)
if(MSVC AND DEFINED BUTTON_MSVC_INCLUDE_PREFIX)
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "${BUTTON_MSVC_INCLUDE_PREFIX}")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
option(BUILD_TESTING "Build independent DesignToCode mapping tests" OFF)
option(D2C_BUILD_MAPPING_COMPILER "Build the existing contract exporter" OFF)
include(CTest)
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Gui Widgets Svg UiTools)
set(KIT "${CMAKE_CURRENT_SOURCE_DIR}/../../ComponentKit")
set(GENERATED_COMPONENT_SDK_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${KIT}" "${CMAKE_BINARY_DIR}/component-kit")
add_library(mapping_core STATIC mapping/QtMapping.cpp mapping/QtMapping.h)
target_include_directories(mapping_core PUBLIC mapping)
target_link_libraries(mapping_core PUBLIC d2c_theme d2c_appearance Qt5::UiTools)
add_library(mapping_demo STATIC ui/MappingPage.cpp ui/MappingPage.h ui/MappingGuide.cpp ui/MappingGuide.h
    mapping/generated/MappingGenerated.cpp mapping/generated/MappingGenerated.h mapping/generated/MappingGenerated.ui)
target_include_directories(mapping_demo PUBLIC ui mapping/generated)
target_link_libraries(mapping_demo PUBLIC mapping_core)
set(RESOURCES mapping-resources.qrc "${KIT}/runtime/component-resources.qrc")
add_executable(qt_meta_scanner mapping/QtMetaScanner.cpp "${KIT}/runtime/component-resources.qrc")
target_link_libraries(qt_meta_scanner PRIVATE d2c_theme d2c_appearance)
add_executable(design_to_code main.cpp ${RESOURCES})
target_link_libraries(design_to_code PRIVATE mapping_demo)
install(TARGETS design_to_code RUNTIME DESTINATION .)
install(FILES Run.cmd DESTINATION .)
install(DIRECTORY "${KIT}/licenses/" DESTINATION licenses)
if(D2C_BUILD_MAPPING_COMPILER)
    add_executable(mapping_compiler mapping/MappingCompiler.cpp ${RESOURCES})
    target_link_libraries(mapping_compiler PRIVATE mapping_core)
endif()
if(BUILD_TESTING)
    find_package(Qt5 5.15 REQUIRED COMPONENTS Test)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    if(NOT TEST_OUTPUT_DIR)
        set(TEST_OUTPUT_DIR "${CMAKE_BINARY_DIR}/reports")
    endif()
    file(MAKE_DIRECTORY "${TEST_OUTPUT_DIR}")
    add_executable(mapping_tests tests/MappingTests.cpp ${RESOURCES})
    target_link_libraries(mapping_tests PRIVATE mapping_demo Qt5::Test)
    target_compile_definitions(mapping_tests PRIVATE TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}" TEST_PYTHON="${Python3_EXECUTABLE}")
    foreach(PROFILE light-comfortable light-compact dark-comfortable dark-compact)
        add_test(NAME mapping_${PROFILE} COMMAND mapping_tests)
        set_tests_properties(mapping_${PROFILE} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;DEMO_PROFILE=${PROFILE}" TIMEOUT 180)
    endforeach()
    foreach(SUITE design_normalization pixso_button independence qt_native_catalog)
        add_test(NAME ${SUITE} COMMAND "${Python3_EXECUTABLE}" -B -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -p "test_${SUITE}.py")
    endforeach()
    add_test(NAME design_evidence_contract COMMAND "${Python3_EXECUTABLE}" -B -m unittest discover
        -s "${CMAKE_CURRENT_SOURCE_DIR}/../../DesignEvidence/tests" -p "test_contract.py")
    add_test(NAME qt_native_meta_catalog COMMAND qt_meta_scanner
        "${CMAKE_CURRENT_SOURCE_DIR}/mapping/qt-native-catalog.json"
        "${TEST_OUTPUT_DIR}/qt-native-meta-report.json")
    set_tests_properties(qt_native_meta_catalog PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 180)
endif()
